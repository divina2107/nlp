# Lowercase Conversion

print(f"Lowercase Tokens: {lowercase_tokens}")

# Punctuation Removal
print("\nPunctuation Removal")
no_punct_tokens = [token for token in lowercase_tokens if token not in string.punctuation]
print(f"Punctuation Removal: {no_punct_tokens}")

# Stop Word Remover
print("\nStop Word Remover")
stop_words = set(stopwords.words('english'))
no_stop_tokens = [token for token in no_punct_tokens if token not in stop_words]
print(f"Tokens without stop words: {no_stop_tokens}")

# Text Correction
print("\nText Correction")
correct_tokens = [str(TextBlob(token).correct()) for token in no_stop_tokens]
print(f"Token after spelling correction: {correct_tokens}")

# Stemming
print("\nStemming")
print(f"Original Text: {text1}")
tokens = word_tokenize(text1)
print(f"Tokens: {tokens}")
stemmer = PorterStemmer()
stemmed_tokens = [stemmer.stem(token) for token in tokens]
print(f"Stemmed Tokens: {stemmed_tokens}")

# Lemmatization
print("\nLemmatization")
print(f"Original Text: {text1}")
lemmatizer = WordNetLemmatizer()
print(f"Tokens: {tokens}")
lemmatized_tokens = [lemmatizer.lemmatize(token) for token in tokens]
print(f"Lemmatized Tokens: {lemmatized_tokens}")

from nltk.tokenize import word_tokenize
def chunk(text):
    words=word_tokenize(text)
    print(f"Words: {words}")
    pos_tags=nltk.pos_tag(words)
    print(f"POS Tags: {pos_tags}")
    grammar="NP: {<DT>?<JJ>*<NN>}"
    chunk_parser=nltk.RegexpParser(grammar)
    tree=chunk_parser.parse(pos_tags)
    print(f"Chunk Tree: {tree}")
    tree.pretty_print()

import nltk
from nltk.corpus import stopwords
nltk.download('stopwords')
nltk.download('punkt_tab')

def generateNGram(text, n=1):
    words=nltk.word_tokenize(text)
    words=[word for word in words if word.lower() not in set(stopwords.words('english'))]
    temp=zip(*[words[i:] for i in range(0,n)])
    ngram=[' '.join(ngram) for ngram in temp]
    return ngram
